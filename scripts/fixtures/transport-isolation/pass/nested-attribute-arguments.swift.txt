public final class PutioSDK {
  public var config: PutioSDKConfig

  func request<T>(as type: T.Type) async throws -> sending T {
    let snapshot = config
    return try await perform(config: snapshot, as: type)
  }

  @concurrent @_implements(Transport, perform(config:as:))
  private func perform<T>(config requestConfig: PutioSDKConfig, as type: T.Type) async throws -> sending T {
    try await execute(requestConfig: requestConfig)
  }

  @concurrent
  private func execute(requestConfig: PutioSDKConfig) async throws -> Data { Data() }
}
